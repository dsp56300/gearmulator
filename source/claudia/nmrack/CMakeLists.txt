# Reuse the existing SHA-256 implementation and header-only peripheral models.
# Rack firmware, machine state and scheduling belong exclusively to nmrack.
find_package(Threads REQUIRED)
add_library(nmrackLib STATIC hardware.cpp ../nmm/nmmLib/nmmrom.cpp ../nmm/nmmLib/nmmpatch.cpp)
target_link_libraries(nmrackLib PUBLIC hardwareLib dsp56kEmu Threads::Threads resample)
target_include_directories(nmrackLib PRIVATE ../nmm)
add_executable(nmrack nmrack.cpp)
target_link_libraries(nmrack PRIVATE nmrackLib)
set_property(TARGET nmrack PROPERTY FOLDER "nmrack")
option(gearmulator_BUILD_NMRACK_PLAYER "Build nmrack audio-device player (requires JUCE)" ON)
if(gearmulator_BUILD_NMRACK_PLAYER AND TARGET juce::juce_audio_devices)
    juce_add_console_app(nmrackPlayer PRODUCT_NAME "nmrack Player")
    target_sources(nmrackPlayer PRIVATE player.cpp)
    target_link_libraries(nmrackPlayer PRIVATE nmrackLib juce::juce_audio_devices juce::juce_events)
    target_compile_definitions(nmrackPlayer PRIVATE JUCE_MODAL_LOOPS_PERMITTED=1 JUCE_WEB_BROWSER=0 JUCE_USE_CURL=0)
elseif(gearmulator_BUILD_NMRACK_PLAYER)
    message(STATUS "nmrackPlayer skipped: enable the project's BUILD_JUCEPLUGIN option for JUCE")
endif()
if(BUILD_TESTING)
    add_executable(nmrackTests tests.cpp)
    target_link_libraries(nmrackTests PRIVATE dsp56kEmu Threads::Threads resample)
    add_test(NAME nmrackTests COMMAND nmrackTests)
    add_executable(nmrackHardwareTests hardwaretests.cpp)
    target_link_libraries(nmrackHardwareTests PRIVATE nmrackLib)
    add_executable(nmrackDacTests dactests.cpp)
    target_link_libraries(nmrackDacTests PRIVATE nmrackLib)
    add_executable(nmrackMidiTests miditests.cpp)
    target_link_libraries(nmrackMidiTests PRIVATE nmrackLib)
    add_executable(nmrackPlaybackTests playbacktests.cpp)
    target_link_libraries(nmrackPlaybackTests PRIVATE nmrackLib)
    set(NMRACK_TEST_FIRMWARE "" CACHE FILEPATH "Optional local Rack ROM for hardware regression tests")
    if(NMRACK_TEST_FIRMWARE)
        add_test(NAME nmrackHardwareTests COMMAND nmrackHardwareTests "${NMRACK_TEST_FIRMWARE}")
        set_tests_properties(nmrackHardwareTests PROPERTIES TIMEOUT 120)
        add_test(NAME nmrackDacTests COMMAND nmrackDacTests "${NMRACK_TEST_FIRMWARE}")
        set_tests_properties(nmrackDacTests PROPERTIES TIMEOUT 120)
        set(NMRACK_TEST_PATCH "${PROJECT_SOURCE_DIR}/nord-micro-modular/patches/FourVoices.pch" CACHE FILEPATH "Local four-voice MIDI test patch")
        if(EXISTS "${NMRACK_TEST_PATCH}")
            add_test(NAME nmrackMidiTests COMMAND nmrackMidiTests "${NMRACK_TEST_FIRMWARE}" "${NMRACK_TEST_PATCH}")
            add_test(NAME nmrackMultiDspMidiTests COMMAND nmrackMidiTests "${NMRACK_TEST_FIRMWARE}" "${NMRACK_TEST_PATCH}" 32)
            add_test(NAME nmrackPlaybackTests COMMAND nmrackPlaybackTests "${NMRACK_TEST_FIRMWARE}" "${NMRACK_TEST_PATCH}")
            set_tests_properties(nmrackMidiTests nmrackMultiDspMidiTests nmrackPlaybackTests PROPERTIES TIMEOUT 120 RUN_SERIAL TRUE)
        endif()
    endif()
endif()
