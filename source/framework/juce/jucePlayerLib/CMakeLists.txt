add_library(jucePlayerLib STATIC
	midiPlayer.cpp midiPlayer.h midiFile.cpp midiFile.h
	audioRouting.cpp audioRouting.h midiInputRouting.cpp midiInputRouting.h)
target_link_libraries(jucePlayerLib PUBLIC synthLib juce_plugin_modules)
target_compile_definitions(jucePlayerLib PUBLIC JUCE_GLOBAL_MODULE_SETTINGS_INCLUDED=1)
if(MSVC)
	target_compile_options(jucePlayerLib PRIVATE /fp:precise)
else()
	target_compile_options(jucePlayerLib PRIVATE -fno-finite-math-only)
endif()
target_include_directories(jucePlayerLib PUBLIC ${CMAKE_CURRENT_SOURCE_DIR}/..)
set_property(TARGET jucePlayerLib PROPERTY FOLDER "Gearmulator")

if(TARGET portmidi)
	add_library(juceMidiPorts STATIC portMidiBridge.cpp portMidiBridge.h)
	target_link_libraries(juceMidiPorts PUBLIC synthLib juce_plugin_modules PRIVATE portmidi)
	target_compile_definitions(juceMidiPorts PUBLIC JUCE_GLOBAL_MODULE_SETTINGS_INCLUDED=1)
	target_include_directories(juceMidiPorts PUBLIC ${CMAKE_CURRENT_SOURCE_DIR}/..)
	set_property(TARGET juceMidiPorts PROPERTY FOLDER "Gearmulator")
endif()

if(BUILD_TESTING)
	foreach(test IN ITEMS midiPlayer audioRouting)
		add_executable(${test}_test tests/${test}_test.cpp)
		target_link_libraries(${test}_test PRIVATE jucePlayerLib)
		target_include_directories(${test}_test PRIVATE ${CMAKE_SOURCE_DIR}/source)
		add_test(NAME ${test} COMMAND ${test}_test)
		set_property(TARGET ${test}_test PROPERTY FOLDER "Tests")
	endforeach()
	target_sources(midiPlayer_test PRIVATE tests/midiFile_test.cpp)
endif()
